    <script>
        const socket = io();

        function loginAdmin() {
            const pwd = document.getElementById('adminPwd').value;
            socket.emit('admin-login', pwd);
        }

        socket.on('admin-auth-success', () => {
            document.getElementById('loginScreen').classList.add('hidden');
            document.getElementById('dashboard').classList.remove('hidden');
        });

        socket.on('admin-auth-error', (msg) => {
            const err = document.getElementById('loginError');
            err.innerText = msg;
            err.classList.remove('hidden');
        });

        socket.on('admin-stats-broadcast', (data) => {
            updateDashboard(data);
        });

        socket.on('admin-stats-update', (data) => {
            updateDashboard(data);
        });

        function updateDashboard(data) {
            document.getElementById('statOnline').innerText = data.onlineCount;
            document.getElementById('statVip').innerText = data.vipCount;

            const tbody = document.getElementById('usersTableBody');
            if (data.users.length === 0) {
                tbody.innerHTML = '<tr><td colspan="5" class="p-4 text-center text-slate-500 italic">Aucun utilisateur connecté.</td></tr>';
                return;
            }

            tbody.innerHTML = '';
            data.users.forEach(user => {
                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-800/50 transition";
                tr.innerHTML = `
                    <td class="p-3 font-bold text-slate-200">${escapeHtml(user.username)}</td>
                    <td class="p-3">${user.isVip ? '<span class="bg-amber-500/20 text-amber-400 px-2 py-0.5 rounded text-[10px] font-bold">VIP</span>' : '<span class="bg-slate-800 text-slate-400 px-2 py-0.5 rounded text-[10px]">Gratuit</span>'}</td>
                    <td class="p-3 text-slate-400 font-mono">${user.ip}</td>
                    <td class="p-3 text-slate-400">${user.connectedAt}</td>
                    <td class="p-3 text-right">
                        <button onclick="banUser('${user.id}')" class="bg-rose-600/20 hover:bg-rose-600 text-rose-400 hover:text-white px-2.5 py-1 rounded transition font-bold text-[10px]">Bannir</button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function banUser(socketId) {
            if (confirm("Voulez-vous vraiment bannir cet utilisateur instantanément ?")) {
                socket.emit('admin-ban-user', socketId);
            }
        }

        function escapeHtml(text) {
            return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
        }
    </script>
